📝 سكربت تحديث الجداول (SQL Script)
لإضافة الحقول الجديدة (القفص، الطفرة، الأحجال) للجدول الموجود:
-- تحديث جدول الأزواج بالحقول الجديدة alter table pairs add column if not exists cage text default '', add column if not exists mutation text default '', add column if not exists status text default 'ready', add column if not exists pairing_date text default '', add column if not exists male_ring text default '', add column if not exists female_ring text default ''; -- تحديث جدول المصروفات بالتصنيفات (علف، أقفاص، بياضات، أدوية، إلخ) alter table expenses add column if not exists category text default 'feed'; -- جدول الزغاليل والشبابيات المستقلة create table if not exists chicks ( id text primary key, name text not null, ring text default '', pair_id text default '', species text default '', mutation text default '', gender text default 'unknown', hatch_date text default '', status text default 'young', cage text default '', price text default '', photo text, notes text default '', created_at timestamptz default now() ); -- جدول سجل المبيعات والإيرادات create table if not exists sales ( id text primary key, date text not null, amount numeric not null, category text default 'chicks', item text default '', buyer text default '', note text default '', created_at timestamptz default now() ); -- إتاحة صلاحية رفع الصور إلى Bucket bird-photos create policy "Public Storage bird-photos" on storage.objects for all using (bucket_id = 'bird-photos') with check (bucket_id = 'bird-photos');
